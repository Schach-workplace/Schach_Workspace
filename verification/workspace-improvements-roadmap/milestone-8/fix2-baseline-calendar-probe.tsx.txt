// @vitest-environment happy-dom
import { render, screen, fireEvent, waitFor, cleanup } from '@testing-library/react';
import { QueryClient, QueryClientProvider } from '@tanstack/react-query';
import { beforeEach, afterEach, it, expect, vi } from 'vitest';
import { CalendarPage } from './CalendarWorkload';
import sample from '../data/sample.json';
import { clearAllDrafts } from '../communication/drafts';
const s = vi.hoisted(() => ({
  w: {} as any,
  urls: [] as string[],
  hold: false,
  release: undefined as (() => void) | undefined,
}));
vi.mock('../store', () => ({ useWorkspace: () => s.w }));
vi.mock('../components/PlanningView', () => ({ default: () => null }));
const tasks = Array.from({ length: 109 }, (_, i) => ({
  id: 'cal-' + i,
  code: 'C-' + i,
  title:
    i === 105
      ? 'Older beyond first100'
      : i === 106
        ? 'Timed local midnight'
        : i === 107
          ? 'Accepted older'
          : 'Calendar work ' + i,
  status: i === 107 ? 'done' : i === 108 ? 'cancelled' : 'todo',
  priority: 'medium',
  progress: 0,
  ownerId: 'u',
  departmentId: '',
  projectId: '',
  dueDate: i === 106 ? '2026-10-10' : '2026-10-11',
  deadlineMode: i === 106 ? 'timed' : 'all_day',
  dueAt: i === 106 ? '2026-10-10T18:45:00.123456Z' : '',
  dueTimezone: i === 106 ? 'Asia/Kolkata' : '',
  version: 1,
}));
beforeEach(() => {
  clearAllDrafts();
  s.urls = [];
  s.hold = false;
  s.release = undefined;
  vi.useFakeTimers({ toFake: ['Date'] });
  vi.setSystemTime(new Date('2026-10-11T06:00:00Z'));
  s.w = {
    mode: 'api',
    companyId: 'c',
    session: { user: { id: 'u' } },
    data: {
      ...sample.workspaces[0],
      readBasis: 'same',
      company: { ...sample.workspaces[0].company, timezone: 'Asia/Kolkata' },
    },
  };
  vi.stubGlobal(
    'fetch',
    vi.fn(async (url) => {
      const path = new URL(String(url), 'http://localhost');
      s.urls.push(path.pathname + path.search);
      if (s.hold)
        await new Promise<void>((r) => {
          s.release = r;
        });
      const all = path.pathname.endsWith('/planning/calendar');
      const rows = all
        ? tasks
            .filter(
              (t) =>
                path.searchParams.get('openOnly') === 'false' ||
                !['done', 'cancelled'].includes(t.status),
            )
            .map((t) => ({ ...t, dueDate: '2026-10-11' }))
        : path.searchParams.has('cursor') ? tasks.slice(100) : tasks.slice(0, 100);
      return new Response(
        JSON.stringify(
          all
            ? { tasks: rows, complete: true, timezone: 'Asia/Kolkata', readBasis: 'current' }
            : { tasks: rows, cursor: path.searchParams.has('cursor') ? '' : 'first100', hasMore: !path.searchParams.has('cursor') },
        ),
        { status: 200 },
      );
    }),
  );
});
afterEach(() => {
  cleanup();
  vi.unstubAllGlobals();
  vi.useRealTimers();
});
const mount = () =>
  render(
    <QueryClientProvider client={new QueryClient()}>
      <CalendarPage onTask={() => {}} />
    </QueryClientProvider>,
  );

it('baseline correctly traverses valid Task pages and shows older/timed/accepted work beyond100', async () => {
  mount();
  await waitFor(() => expect(screen.getByText('107 deadlines on this day')).toBeTruthy());
  expect(screen.getByText('Older beyond first100')).toBeTruthy();
  expect(screen.getByText('Timed local midnight')).toBeTruthy();
  expect(s.urls).toEqual(['/api/companies/c/tasks', '/api/companies/c/tasks?cursor=first100']);
  fireEvent.click(screen.getByLabelText('Open work only'));
  expect(screen.getByText('109 deadlines on this day')).toBeTruthy();
  expect(screen.getByText('Accepted older')).toBeTruthy();
  console.log(JSON.stringify({ baseline:'0ad8176',pages:s.urls.length,open107:true,all109:true,older:true,timedLocal:true,accepted:true }));
});
it('bounds source work to the complete selected range instead of aggregating10001 outside-range tasks', async () => {
  let fetched=0, bytes=0, calls=0;
  const outside=Array.from({length:10001},(_,i)=>({...tasks[0],id:'outside-'+i,title:'Outside '.repeat(200),dueDate:'2020-01-01'}));
  const selected={...tasks[105],title:'Selected date work'};
  const source=[...outside,selected];
  vi.stubGlobal('fetch',vi.fn(async url=>{
    const path=new URL(String(url),'http://localhost');calls++;
    const offset=Number(path.searchParams.get('cursor')||0);
    const rows=source.slice(offset,offset+100);
    fetched+=rows.length;
    const more=offset+rows.length<source.length;
    const body=JSON.stringify({tasks:rows,hasMore:more,cursor:more?String(offset+rows.length):''});bytes+=body.length;
    return new Response(body,{status:200});
  }));
  mount();await waitFor(()=>expect(screen.getByText('1 deadlines on this day')).toBeTruthy(),{timeout:15000});
  console.log(JSON.stringify({baseline:'0ad8176',calls,fetched,bytes,visibleSelectedTasks:1,taskReadCurrentDateFilter:false}));
  expect(fetched).toBeLessThanOrEqual(10000);
  expect(bytes).toBeLessThanOrEqual(8<<20);
},20000);
